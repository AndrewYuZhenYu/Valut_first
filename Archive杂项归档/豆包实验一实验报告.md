```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from datetime import datetime

# 设置中文字体
plt.rcParams['font.sans-serif'] = ['WenQuanYi Zen Hei']
plt.rcParams['axes.unicode_minus'] = False
plt.rcParams['figure.figsize'] = (16, 12)

# 1. 读取三个城市的填充后数据
def load_city_data(file_path, city_name):
    """读取城市温度数据并添加城市标识"""
    df = pd.read_csv(file_path)
    # 转换Time列为datetime格式
    df['Time'] = pd.to_datetime(df['Time'], format='%Y%m%d%H')
    # 添加城市名称列
    df['City'] = city_name
    # 提取日期和小时信息（用于后续分析）
    df['Date'] = df['Time'].dt.date
    df['Hour'] = df['Time'].dt.hour
    return df

# 读取数据
beijing_df = load_city_data('/mnt/Beijing_Filled.csv', '北京')
guangzhou_df = load_city_data('/mnt/Guangzhou_Filled.csv', '广州')
nanjing_df = load_city_data('/mnt/Nanjing_filled (1).csv', '南京')

# 合并所有数据
all_cities_df = pd.concat([beijing_df, guangzhou_df, nanjing_df], ignore_index=True)

# 2. 计算关键统计量
city_stats = all_cities_df.groupby('City')['TMP'].agg([
    'count', 'mean', 'std', 'min', 'max',
    lambda x: x.quantile(0.25),  # 25%分位数
    lambda x: x.quantile(0.5),   # 中位数（50%分位数）
    lambda x: x.quantile(0.75)   # 75%分位数
]).round(2)

# 重命名列
city_stats.columns = ['数据点数', '平均值(℃)', '标准差(℃)', '最小值(℃)', '最大值(℃)', 
                     '25%分位数(℃)', '中位数(℃)', '75%分位数(℃)']

# 计算额外统计量：温度极差（max-min）
city_stats['温度极差(℃)'] = (city_stats['最大值(℃)'] - city_stats['最小值(℃)']).round(2)

# 按平均值降序排列（广州>南京>北京）
city_stats = city_stats.sort_values('平均值(℃)', ascending=False)

print("=== 三城市温度数据关键统计量对比 ===")
print(city_stats)

# 3. 保存统计量表格（便于后续引用）
stats_table = city_stats.copy()
```

    1. 早晨 6-12 时升温速率最快，三个城市每小时升温约 0.8-1.2℃

## 五、综合总结与结论

### 5.1 核心结论

1. **地域气候差异显著**：三个城市温度水平呈现明显的南北梯度，广州（亚热带温和）> 南京（温带微凉）> 北京（温带寒冷），符合中国东部季风气候的纬度分异规律。
    
2. **稳定性排序**：广州温度最稳定（标准差 3.45℃），南京次之（4.84℃），北京最不稳定（4.76℃），反映了南方沿海气候比北方内陆更温和湿润的特点。
    
3. **风险特征**：
    
    1. 北京存在严寒风险（最低 - 10℃），需防范低温冻害
        
    2. 南京无严寒但有霜冻风险（最低 - 3℃）
        
    3. 广州无低温风险，冬季气候宜人
        
4. **时间规律一致性**：三个城市在日尺度（昼夜变化）、周尺度（降温 - 升温周期）均表现出同步性，说明区域天气系统对温度的主导作用。
